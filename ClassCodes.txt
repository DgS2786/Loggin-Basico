import React, { useState } from 'react';
import { View, Text, Button, StyleSheet } from 'react-native';

const App = () => {
  console.log('Rendering App');

  const [count, setCount] = useState(0);
  const [countTen, setCountTen] = useState(0);
  const [countHund, setCountHund] = useState(0);

  const handleButtonClickListener = () => {
    setCount(count + 1);
    setCountTen(countTen + 10);
    setCountHund(countHund + 100);
  };

  return (
    <View style={styles.main}>
      <Text>Sample App</Text>
      <Button onPress={handleButtonClickListener} title="Click Me" />
      <Text>Count One: {count}</Text>
      <Text>Count Ten: {countTen}</Text>
      <Text>Count Hund: {countHund}</Text>
    </View>
  );
};

const styles = StyleSheet.create({
  main: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center'
  }
});

export default App;
